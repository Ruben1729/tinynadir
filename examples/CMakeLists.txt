function(tinynadir_example name source)
    foreach(_module IN LISTS ARGN)
        if(NOT TARGET ${_module}::${_module})
            return()
        endif()
    endforeach()
    add_executable(tinynadir_${name} ${source})
    set_target_properties(tinynadir_${name} PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    foreach(_module IN LISTS ARGN)
        target_link_libraries(tinynadir_${name} PRIVATE ${_module}::${_module})
    endforeach()
    if(TINYNADIR_BUILD_TESTS)
        add_test(NAME tinynadir.example.${name} COMMAND tinynadir_${name})
    endif()
endfunction()
tinynadir_example(detumble detumble.c tinybdot)
tinynadir_example(wheel_pointing wheel_pointing.c tinypd tinywheels)
tinynadir_example(snapshot_attitude snapshot_attitude.c tinyomm tinytime tinyframes tinysun tinyigrf tinyrotation tinytriad)
